{# ---- combo chart: bars (actual vs forecast) + line on a secondary axis ---- Deck slide 3 (docs/ammn-slides/slide3-visual-spec.md). The bars carry the absolute series and the line the ratio; the first `actual_n` bars are solid navy and the rest are drawn lighter with a hatch overlay, so actual vs projection is readable without reading the axis labels. #} {% macro svg_combo(uid, labels, bars, line, actual_n, bars_fmt, line_fmt, bar_unit, line_unit, w=248, h=152) -%} {% set pad_l = 34 %}{% set pad_r = 38 %}{% set pad_t = 15 %}{% set pad_b = 22 %} {% set cw = w - pad_l - pad_r %}{% set ch = h - pad_t - pad_b %} {% set n = labels | length %} {% set bvals = bars | select('ne', None) | list %} {% set bmax = (bvals | max) if bvals else 1 %} {% set lvals = line | select('ne', None) | list %} {% set lmax = (lvals | max) if lvals else 1 %} {% set lmin = (lvals | min) if lvals else 0 %} {# The line's scale always contains zero: without that, a negative year draws at the same height as a small positive one and only its label says which is which (the spec forbids exactly that). #} {% set l_lo = [lmin, 0] | min %} {% set l_hi = [lmax, 0] | max %} {% set lspan = (l_hi - l_lo) if (l_hi - l_lo) > 0 else 1 %} {% set step = cw / n if n else cw %} {% set bw = step * 0.56 %} {% if not labels or not bvals %} {# No series in the payload: draw the empty frame and say so rather than crash the render. #} {% else %} {% endif %} {%- endmacro %}
